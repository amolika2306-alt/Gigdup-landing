  var pS=document.getElementById('panel-s'), pB=document.getElementById('panel-b');
  function show(isStudent){
    tabS.setAttribute('aria-selected',isStudent); tabB.setAttribute('aria-selected',!isStudent);
    tabS.tabIndex=isStudent?0:-1; tabB.tabIndex=isStudent?-1:0;
    pS.hidden=!isStudent; pB.hidden=isStudent;
  }
  tabS.addEventListener('click',function(){show(true)});
  tabB.addEventListener('click',function(){show(false)});
  [tabS,tabB].forEach(function(t){
    t.addEventListener('keydown',function(e){
      if(e.key==='ArrowRight'||e.key==='ArrowLeft'){ var s=tabS.getAttribute('aria-selected')==='true'; show(!s); (s?tabB:tabS).focus(); }
    });
  });

  // CTA buttons pre-select the role in the form
  document.querySelectorAll('[data-role]').forEach(function(a){
    a.addEventListener('click',function(){ document.getElementById('role').value=a.getAttribute('data-role'); });
  });

  // Form: front-end only. Replace this handler with a real endpoint
  // (for example Formspree, Mailchimp, or your own API) before launch.
